Ejemplo:
CREATE OR REPLACE PROCEDURE MOSTRAR_DATOS_EMPLEADO(V_ID IN NUMBER)
IS
    V_NOMBRE VARCHAR2(200);
    V_SUELDO NUMBER;
BEGIN
    SELECT FIRST_NAME ||' '||LAST_NAME, SALARY
    INTO V_NOMBRE, V_SUELDO
    FROM EMPLOYEES
    WHERE EMPLOYEE_ID = V_ID;
    
    DBMS_OUTPUT.PUT_LINE('****************************');
    DBMS_OUTPUT.PUT_LINE('Nombre: '||V_NOMBRE);
    DBMS_OUTPUT.PUT_LINE('Sueldo: $'||V_SUELDO);
    DBMS_OUTPUT.PUT_LINE('****************************');
END;

/*
    CREAR UN PROCEDIMIENTO ALMACENADO QUE PERMITA RECIBIR COMO ARGUMENTO
    EL ID DE UN EMPLEADO Y EL PORCENTAJE D SUELDO QUE SE LE AUMENTARÁ
*/

CREATE OR REPLACE PROCEDURE PR_AUMENTO_SUELDO(V_ID_EMPLEADO IN NUMBER, V_PORCENTAJE IN NUMBER)
IS
    V_PORC_INVALIDO EXCEPTION;
BEGIN
    IF V_PORCENTAJE < 0 OR V_PORCENTAJE > 100 THEN
        RAISE V_PORC_INVALIDO;
    END IF;

    UPDATE EMPLEADOS
    SET SALARY = SALARY + (SALARY * V_PORCENTAJE / 100)
    WHERE EMPLOYEE_ID = V_ID_EMPLEADO;
EXCEPTION
    WHEN V_PORC_INVALIDO THEN
        DBMS_OUTPUT.PUT_LINE('ERROR: EL PORCENAJE DEBE ESTAR ENTRE EL 0 Y 100');
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('ERROR: EL EMPLEADO CON ID '||V_ID_EMPLEADO||' NO ESTA REGISTRADO EN LA BASE DE DATOS');
END;

EXEC PR_AUMENTO_SUELDO(150, 15);
