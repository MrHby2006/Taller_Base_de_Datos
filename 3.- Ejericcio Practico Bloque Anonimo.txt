/*
    CREA UN BLOQUE ANONIMO QUE PERMITA RECALCULAR EL SUELDO DE LOS EMPLEADOS DE UN MISMO DEPARTAMENTO.
    - SE DEBE INGRESAR EL ID DEL DEPARTAMENTO COMO PARAMETRO AL EJECUTAR EL BLOQUE ANONIMO.
    - SE DEBE ANALIZAR EL SUELDO DE LOS EMPLEADOS SEGUN LOS SIGUIENTES CRITERIOS: 
        1.- SI TIENE UN SUELDO SUPERIOR A 4.000 SE DEBE AUMENTAR EL SUELDO EN UN 4%
        2.- SI EL SUELDO DEL EMPLEADO ES 4.000 O INFERIOR EL SUELDO DEBE AUMENTAR EN UN 10%
    - LOS DATOS Y CALCULOS GENERADOS DEBEN GUARDARSE EN LA TABLA AJUSTES_SALARIALES:
        CREATE TABLE AJUSTES_SALARIALES(
            COD_AJUSTE       NUMBER GENERATED BY DEFAULT AS IDENTITY,
            EMPLOYEE_ID      NUMBER,
            NOMBRE           VARCHAR(100),
            DEPARTAMENTO     VARCHAR(50),
            PAIS             VARCHAR(50),
            SUELDO_ORIGINAL  NUMBER(10, 2),
            PORCENTAJE       NUMBER(10, 2),
            SUELDO_NUEVO     NUMBER(10, 2),
            FECHA_PROCESO    DATE
        );
*/
DROP TABLE AJUSTES_SALARIALES;

CREATE TABLE AJUSTES_SALARIALES(
    COD_AJUSTE       NUMBER GENERATED BY DEFAULT AS IDENTITY,
    EMPLOYEE_ID      NUMBER,
    NOMBRE           VARCHAR(100),
    DEPARTAMENTO     VARCHAR(50),
    PAIS             VARCHAR(50),
    SUELDO_ORIGINAL  NUMBER(10, 2),
    PORCENTAJE       NUMBER(10, 2),
    SUELDO_NUEVO     NUMBER(10, 2),
    FECHA_PROCESO    DATE
);

SELECT E.EMPLOYEE_ID AS ID_EMPLEADO,
       E.FIRST_NAME||' '||E.LAST_NAME AS NOMBRE,
       D.DEPARTMENT_NAME AS DEPARTAMENTO,
       C.COUNTRY_NAME AS PAIS,
       E.SALARY AS SUELDO_ORIGINAL
FROM EMPLOYEES E
LEFT JOIN DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
LEFT JOIN LOCATIONS L ON D.LOCATION_ID = L.LOCATION_ID
LEFT JOIN COUNTRIES C ON L.COUNTRY_ID = C.COUNTRY_ID;

DECLARE
    V_DEPARTAMENTO NUMBER := &ID;
    V_PORCENTAJE      NUMBER(10, 2);
    V_SUELDO_NUEVO    NUMBER(10, 2);
BEGIN
    FOR R_AJUSTES IN (SELECT E.EMPLOYEE_ID AS ID_EMPLEADO,
                             E.FIRST_NAME||' '||E.LAST_NAME AS NOMBRE,
                             D.DEPARTMENT_NAME AS DEPARTAMENTO,
                             C.COUNTRY_NAME AS PAIS,
                             E.SALARY AS SUELDO_ORIGINAL
                      FROM EMPLOYEES E
                      LEFT JOIN DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
                      LEFT JOIN LOCATIONS L ON D.LOCATION_ID = L.LOCATION_ID
                      LEFT JOIN COUNTRIES C ON L.COUNTRY_ID = C.COUNTRY_ID
                      WHERE D.DEPARTMENT_ID = V_DEPARTAMENTO)
    LOOP
        IF R_AJUSTES.SUELDO_ORIGINAL > 4000 THEN
            V_PORCENTAJE := 4;
            V_SUELDO_NUEVO := R_AJUSTES.SUELDO_ORIGINAL * 1.04;
        ELSIF R_AJUSTES.SUELDO_ORIGINAL <= 4000 THEN
            V_PORCENTAJE := 10;
            V_SUELDO_NUEVO := R_AJUSTES.SUELDO_ORIGINAL * 1.10;
        END IF;
        INSERT INTO AJUSTES_SALARIALES(EMPLOYEE_ID, NOMBRE, DEPARTAMENTO, PAIS, SUELDO_ORIGINAL, PORCENTAJE, SUELDO_NUEVO, FECHA_PROCESO)
        VALUES (R_AJUSTES.ID_EMPLEADO, R_AJUSTES.NOMBRE, R_AJUSTES.DEPARTAMENTO, R_AJUSTES.PAIS, R_AJUSTES.SUELDO_ORIGINAL, V_PORCENTAJE, V_SUELDO_NUEVO, SYSDATE);
    END LOOP;
END;
