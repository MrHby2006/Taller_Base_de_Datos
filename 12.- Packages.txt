CREATE OR REPLACE PACKAGE GESTION_EMPLEADOS AS
    FUNCTION CAL_SUELDO_ANUAL(V_ID_EMPLE IN NUMBER) RETURN NUMBER;
    
    PROCEDURE ACTUALIZAR_SUELDO(V_ID_EMPLE IN NUMBER, V_NUEVO_SUELDO IN NUMBER);
END GESTION_EMPLEADOS;

CREATE OR REPLACE PACKAGE BODY GESTION_EMPLEADOS AS
    FUNCTION CAL_SUELDO_ANUAL(V_ID_EMPLE IN NUMBER) RETURN NUMBER
    IS
        V_SUELDO NUMBER;
        V_SALARIO_ANUAL NUMBER;
    BEGIN
        SELECT SALARY
        INTO V_SUELDO
        FROM EMPLOYEES
        WHERE EMPLOYEE_ID = V_ID_EMPLE;
        
        V_SALARIO_ANUAL := V_SUELDO * 12;                    
        RETURN V_SALARIO_ANUAL;
    END CAL_SUELDO_ANUAL;
    
    PROCEDURE ACTUALIZAR_SUELDO(V_ID_EMPLE IN NUMBER, V_NUEVO_SUELDO IN NUMBER)
    IS
    BEGIN
        IF V_NUEVO_SUELDO <= 0 THEN
            RAISE_APPLICATION_ERROR(-20001, 'EL SUELDO DEBE SER MAYOR A 0');
        END IF;
    
        UPDATE EMPLOYEES
        SET SALARY = V_NUEVO_SUELDO
        WHERE EMPLOYEE_ID = V_ID_EMPLE;
    END ACTUALIZAR_SUELDO;
END GESTION_EMPLEADOS;

SELECT FIRST_NAME,
       EMPLOYEE_ID,
       SALARY,
       GESTION_EMPLEADOS.CAL_SUELDO_ANUAL(EMPLOYEE_ID) AS SUELDO_ANUAL
FROM EMPLOYEES;
