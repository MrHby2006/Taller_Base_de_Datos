DECLARE
    R_DEPARTAMENTO DEPARTMENTS%ROWTYPE;
BEGIN
    SELECT *
    INTO R_DEPARTAMENTO
    FROM DEPARTMENTS
    WHERE DEPARTMENT_ID = 50;
    
    DBMS_OUTPUT.PUL_LINE(R_DEPARTAMENTO.NOMBRE_DEPA);
END;

/*
    CREE UN VARRAY QUE PERMITA GUARDAR TODOS LOS NOMBRES DE LOS
    DEPARTAMENTOS DE LA EMPRESA
    
    Y MUESTRELSO AL FINAL
*/
DECLARE
    TYPE VARRAY_DEPAS IS VARRAY(27)
    OF VARCHAR(50);
    
    VR_DEPAS VARRAY_DEPAS := VARRAY_DEPAS();
BEGIN
    FOR R_DEPARTAMENTOS IN (SELECT DEPARTMENT_NAME FROM DEPARTMENTS)
    LOOP 
        VR_DEPAS.EXTEND;
        VR_DEPAS(VR_DEPAS.LAST) := R_DEPARTAMENTOS.DEPARTMENT_NAME;
    END LOOP;
    FOR i IN 1..VR_DEPAS.LAST
    LOOP
        DBMS_OUTPUT.PUT_LINE(VR_DEPAS(i));
    END LOOP;
END;
