CREATE OR REPLACE FUNCTION CALCULAR_BONO(V_SUELDO NUMBER) --SE CREA O SE REEMPLAZA EN CASO DE QUE EXISTA, ASI SE CREA UNA FUNCION
RETURN NUMBER
IS
    V_BONO NUMBER;
BEGIN
    IF V_SUELDO < 5000 THEN
        V_BONO := V_SUELDO * 0.15;
    ELSIF V_SUELDO BETWEEN 5000 AND 1000 THEN
        V_BONO := V_SUELDO * 0.1;
    ELSIF V_SUELDO > 1000 THEN
        V_BONO := V_SUELDO * 0.05;
    END IF;
    
    RETURN V_BONO;
END;

SELECT FIRST_NAME||' '||LAST_NAME,
       SALARY,
       JOB_ID,
       CALCULAR_BONO(SALARY) --SE LLAMA A LA FUNCION YA CREADA
FROM EMPLOYEES;

-------------------------------------------------------------------------
CREATE OR REPLACE PROCEDURE MOSTRAR_BONO(V_ID NUMBER) --PARA CREAR PROCEDIMIENTOS, NO NECESARIAMENTE RETORNAN ALGO
IS
    V_NOMBRE VARCHAR2(50);
    V_SUELDO NUMBER;
    V_BONO NUMBER;
    V_TOTAL NUMBER;
BEGIN
    SELECT FIRST_NAME||' '||LAST_NAME,
           SALARY,
           CALCULAR_BONO(SALARY)
    INTO V_NOMBRE,
         V_SUELDO,
         V_BONO
    FROM EMPLOYEES
    WHERE EMPLOYEE_ID = V_ID;
    
    V_TOTAL := V_SUELDO + V_BONO;
    
    DBMS_OUTPUT.PUT_LINE('Nombre del empleado: '||V_NOMBRE);
    DBMS_OUTPUT.PUT_LINE('Su sueldo es de: $'||V_SUELDO);
    DBMS_OUTPUT.PUT_LINE('Su bono es de: $'||V_BONO);
    DBMS_OUTPUT.PUT_LINE('El total de haberes del empleado es de: $'||V_TOTAL);

EXCEPTION -- AÃ‘ADIMOS LAS EXEPCIONES
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('El empleado con id '||V_ID||' no existe');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Ocurrio un error');
END;

-------------------------------------------------------------------------
EXEC MOSTRAR_BONOS(100); --PARA USAR PROCEDIMIENTOS CREADOS
