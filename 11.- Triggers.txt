CREATE OR REPLACE TRIGGER TRG_VALIDAR_SUELDO
BEFORE
UPDATE
ON EMPLEADOS
FOR EACH ROW
BEGIN
    IF :NEW.SALARY <= 0 THEN
        RAISE_APPLICATION_ERROR(-20001, 'EL SUELDO DEBE SER MAYOR A 0');
    ELSE
        DBMS_OUTPUT.PUT_LINE('EL TRIGGER SE DISPARO PERO EL SUELDO ES VÁLIDO');
    END IF;
END;

UPDATE EMPLEADOS
SET SALARY = 22000
WHERE EMPLOYEE_ID = 100;

UPDATE EMPLEADOS
SET SALARY = -1500
WHERE EMPLOYEE_ID = 100;

CREATE TABLE AUDITORIA_SUELDOS(
    ID_AUDITORIA NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    EMPLOYEE_ID NUMBER,
    SUELDO_ANTERIOS NUMBER,
    SUELDO_NUEVO NUMBER,
    FECHA_CAMBIO DATE
);

CREATE OR REPLACE TRIGGER TRG_AUDITORIA_SUELDOS
AFTER UPDATE OF SALARY ON EMPLEADOS //APUNTAMOS A UNA COLUMNA EN ESPECIFICO PARA EVITAR ERRORES
FOR EACH ROW
BEGIN
    INSERT INTO AUDITORIA_SUELDO(
        EMPLOYEE_ID,
        SUELDO_ANTERIOR,
        SUELDO_NUEVO,
        FECHA_CAMBIO
    )VALUES(
        :NEW.EMPLOYEE_ID,
        :OLD.SALARY,
        :NEW.SALARY,
        SYSDATE
    );
END;
