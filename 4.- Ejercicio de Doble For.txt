SELECT DEPARTMENT_NAME AS DEPA,
       DEPARTMENT_ID AS ID
FROM DEPARTMENTS
ORDER BY DEPARTMENT_ID ASC;

SELECT E.FIRST_NAME||' '||E.LAST_NAME AS NOMBRE
FROM EMPLOYEES E
WHERE E.DEPARTMENT_ID = R_DEPARTAMENTOS.ID;

DECLARE
    V_ENUMERACION NUMBER;
    V_CONTADOR NUMBER;
BEGIN
    FOR R_DEPARTAMENTOS IN (SELECT DEPARTMENT_NAME AS DEPA,
                                   DEPARTMENT_ID AS ID
                            FROM DEPARTMENTS
                            ORDER BY DEPARTMENT_ID ASC)
    LOOP
        V_ENUMERACION := 1;
        V_CONTADOR := 0;
        DBMS_OUTPUT.PUT_LINE(' ');
        DBMS_OUTPUT.PUT_LINE('-----------------------');
        DBMS_OUTPUT.PUT_LINE(R_DEPARTAMENTOS.DEPA);
        DBMS_OUTPUT.PUT_LINE('-----------------------');
        FOR R_EMPLEADOS IN (SELECT E.FIRST_NAME||' '||E.LAST_NAME AS NOMBRE
                            FROM EMPLOYEES E
                            WHERE E.DEPARTMENT_ID = R_DEPARTAMENTOS.ID)
        LOOP
            DBMS_OUTPUT.PUT_LINE(V_ENUMERACION||'.- '||R_EMPLEADOS.NOMBRE);
            V_ENUMERACION := V_ENUMERACION + 1;
            V_CONTADOR := V_CONTADOR + 1;
        END LOOP;
        DBMS_OUTPUT.PUT_LINE(' ');
        DBMS_OUTPUT.PUT_LINE('CANTIDAD DE TRABAJADORES EN LA SECCION DE '||R_DEPARTAMENTOS.DEPA||': '||V_CONTADOR);
    END LOOP;
END;
