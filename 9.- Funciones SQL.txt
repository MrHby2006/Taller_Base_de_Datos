CREATE OR REPLACE FUNCTION ANIOS_ANTIGUEDAD(V_FECHA_CONTRATA IN DATE)
RETURN INTEGER
IS
    V_ANTIGUEDAD INTEGER;
BEGIN
    V_ANTIGUEDAD := TRUNC((SYSDATE - V_FECHA_CONTRATA)/365);
    
    RETURN V_ANTIGUEDAD;
END;

SELECT EMPLOYEE_ID,
       HIRE_DATE,
       ANIOS_ANTIGUEDAD(HIRE_DATE) ANTIGUEDAD
FROM EMPLOYEES;

-- FUNCION QUE PERMITA FORMATEAR EL NOMBRE, RETORNANDO EL
-- VALOR CON LA PRIMERA LETRA EN MAYUSCULA Y ELIMINANDO
-- LOS ESPACIOS VACIOS

CREATE OR REPLACE FUNCTION FORMATEO_NOMBRE(V_NOMBRE IN VARCHAR2)
RETURN VARCHAR2
IS
    V_NOMBRE_FORMATEADO VARCHAR2(50);
BEGIN
    V_NOMBRE_FORMATEADO := TRIM(INITCAP(V_NOMBRE));
    
    RETURN V_NOMBRE_FORMATEADO;
END;

SELECT EMPLOYEE_ID,
       HIRE_DATE,
       ANIOS_ANTIGUEDAD(HIRE_DATE) ANTIGUEDAD,
       FORMATEO_NOMBRE(FIRST_NAME) NOMBRE
FROM EMPLOYEES;

/*
    CREE UNA FUNCION QUE PERMITA RECIBIR COMO ARGUMENTO
    EL SUELDO DE UN EMPLEADO Y SEGUN LA LOGICA DE LA EMPRESA
    PUEDA CATEGORIZARLO Y RETORNAR LA CATEGORIA DEL EMPLEADO:
    SUELDO MENOR QUE 3000 : SUELDO INICIAL
    SUELDO ENTRE 3000 Y 8000 : SUELDO MEDIO
    SUELDO SUPERIOR A 8000 : SUELDO AVANZADO
*/

CREATE OR REPLACE FUNCTION CATEGORIA_SUELDO(V_SUELDO IN NUMBER)
RETURN VARCHAR2
IS
    V_CATEGORIA VARCHAR2(15);
BEGIN
    IF V_SUELDO IS NULL THEN
        V_CATEGORIA := 'SUELDO NO INFORMADO';
    ELSIF V_SUELDO < 3000 THEN
        V_CATEGORIA := 'SUELDO INICIAL';
    ELSIF V_SUELDO BETWEEN 3000 AND 8000 THEN
        V_CATEGORIA := 'SUELDO MEDIO';
    ELSIF V_SUELDO > 8000 THEN
        V_CATEGORIA := 'SUELDO AVANZADO';
    END IF;
    
    RETURN V_CATEGORIA;
END;

SELECT FORMATEO_NOMBRE(FIRST_NAME) NOMBRE,
       SALARY,
       CATEGORIA_SUELDO(SALARY) CATEGORIA
FROM EMPLOYEES;

/*
    CREE UNA FUNCION QUE RECIBA COMO ARGUMENTO EL ID DE UN
    DEPARTAMENTO Y RETORNE E NOMBRE DEL JEFE DEL DEPARTAMENTO
*/

CREATE OR REPLACE FUNCTION JEFE_DEPARTAMENTO(V_ID_DEPARTAMENTO IN NUMBER)
RETURN VARCHAR2
IS
    V_NOMBRE_JEFE VARCHAR2(50);
BEGIN
    SELECT E.FIRST_NAME||' '||E.LAST_NAME
    INTO V_NOMBRE_JEFE
    FROM DEPARTMENTS D
    JOIN EMPLOYEES E ON D.MANAGER_ID = E.EMPLOYEE_ID
    WHERE D.DEPARTMENT_ID = V_ID_DEPARTAMENTO;
    
    IF V_NOMBRE_JEFE IS NULL THEN
        V_NOMBRE_JEFE := 'Departamento Sin Manager';
    END IF;
    
    RETURN V_NOMBRE_JEFE;
END;

SELECT DEPARTMENT_NAME,
       JEFE_DEPARTAMENTO(DEPARTMENT_ID) MANAGER
FROM DEPARTMENTS;
