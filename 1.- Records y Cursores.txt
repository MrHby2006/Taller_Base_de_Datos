DECLARE
    TYPE RECORD_EMPLEADO IS RECORD(
        ID EMPLOYEES.EMPLOYEE_ID%TYPE,
        NOMBRE EMPLOYEES.FIRST_NAME%TYPE,
        APELLIDO EMPLOYEES.LAST_NAME%TYPE,
        SUELDO EMPLOYEES.SALARY%TYPE
    );
    
    R_EMPLEADO      RECORD_EMPLEADO;
    V_NUEVO_SUELDO  NUMBER;
BEGIN
    SELECT EMPLOYEE_ID, FIRST_NAME, LAST_NAME, SALARY
    INTO R_EMPLEADO
    FROM EMPLOYEES
    WHERE EMPLOYEE_ID = 100;
    
    V_NUEVO_SUELDO := R_EMPLEADO.SUELDO *1.1;
    
    DBMS_OUTPUT.PUT_LINE('EL EMPLEADO '||R_EMPLEADO.NOMBRE|| R_EMPLEADO.APELLIDO);
    DBMS_OUTPUT.PUT_LINE('TENIA UN SUELDO DE: $'||R_EMPLEADO.SUELDO);
    DBMS_OUTPUT.PUT_LINE('CON AUMENTO DEL 10% QUEDO EN: $'||V_NUEVO_SUELDO);
END;

DECLARE
    CURSOR C_EMPLEADO IS 
        SELECT EMPLOYEE_ID, FIRST_NAME, LAST_NAME, SALARY
        FROM EMPLOYEES
        WHERE DEPARTMENT_ID = 50;
    
    TYPE RECORD_EMPLEADO IS RECORD(
        ID EMPLOYEES.EMPLOYEE_ID%TYPE,
        NOMBRE EMPLOYEES.FIRST_NAME%TYPE,
        APELLIDO EMPLOYEES.LAST_NAME%TYPE,
        SUELDO EMPLOYEES.SALARY%TYPE
    );
    
    R_EMPLEADO      RECORD_EMPLEADO;
    V_NUEVO_SUELDO  NUMBER;
BEGIN
    OPEN C_EMPLEADO; --EJECUTA LA QUERY (PARA ESO SIRVE EL OPEN)
    LOOP
        FETCH C_EMPLEADO INTO R_EMPLEADO; --RECUPERA LOS DATOS DEL CURSOR (PARA ESO SIRVE EL FETCH) 
        
        EXIT WHEN C_EMPLEADO%NOTFOUND; --PARA SALIR CON CONDICIÃ“N, EN ESTE CASO CUANDO NO SE ENCUENTRE NADA
        
        V_NUEVO_SUELDO := R_EMPLEADO.SUELDO *1.1;
    
        DBMS_OUTPUT.PUT_LINE('EL EMPLEADO '||R_EMPLEADO.NOMBRE|| R_EMPLEADO.APELLIDO);
        DBMS_OUTPUT.PUT_LINE('TENIA UN SUELDO DE: $'||R_EMPLEADO.SUELDO);
        DBMS_OUTPUT.PUT_LINE('CON AUMENTO DEL 10% QUEDO EN: $'||V_NUEVO_SUELDO);
    END LOOP;
    CLOSE C_EMPLEADO;
END;
