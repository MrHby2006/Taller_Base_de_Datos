DECLARE
    V_NOMBRE         EMPLOYEES.FIRST_NAME%TYPE;
    V_ID_EMPLEADO    EMPLOYEES.EMPLOYEE_ID%TYPE := &ID_EMPLEADO;
BEGIN
    SELECT FIRST_NAME
    INTO V_NOMBRE
    FROM EMPLOYEES
    WHERE EMPLOYEE_ID = V_ID_EMPLEADO;
    
    DBMS_OUTPUT.PUT_LINE('EL NOMBRE DEL EMPLEADO ES: '||V_NOMBRE);
EXCEPTION
    WHEN NO_DATA_FOUND THEN -- NO SE ENCONTRO INFORMACION
        DBMS_OUTPUT.PUT_LINE('EL EMPLEADO CON ID '||V_ID_EMPLEADO||' NO EXISTE');
END;

--------------------------------------
DECLARE
    V_NOMBRE         EMPLOYEES.FIRST_NAME%TYPE;
    V_ID_DEPA        NUMBER := &ID_EMPLEADO;
BEGIN
    SELECT FIRST_NAME
    INTO V_NOMBRE
    FROM EMPLOYEES
    WHERE DEPARTMENT_ID = V_ID_DEPA;
    
    DBMS_OUTPUT.PUT_LINE('EL NOMBRE DEL EMPLEADO ES: '||V_NOMBRE);
EXCEPTION
    WHEN TOO_MANY_ROWS THEN
        DBMS_OUTPUT.PUT_LINE('EL DEPARTAMENTO TIENE MAS DE UN EMPLEADO');
    WHEN NO_DATA_FOUND THEN -- MUCHAS FILAS PARA SOLO RETORNAR UNA COSA
        DBMS_OUTPUT.PUT_LINE('EL DEPARTAMENTO NO EXISTE');
    WHEN OTHERS THEN -- EN CASO DE CUALQUIER COSA, SIEMPRE PONERLO AL FINAL
        DBMS_OUTPUT.PUT_LINE('OCURRIO UN ERROR INESPERADO');
END;

DECLARE
    V_NUM1 NUMBER := &NUM1;
    V_NUM2 NUMBER := &NUM2;
    
    V_RESULTADO NUMBER;
BEGIN
    V_RESULTADO := V_NUM1 / V_NUM2;
    
    DBMS_OUTPUT.PUT_LINE('EL RESULTADO ES: '||V_RESULTADO);
EXCEPTION
    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE('NO SE PUEDE DIVIDIR POR 0');
END;

DECLARE
    V_NOMBRE         EMPLOYEES.FIRST_NAME%TYPE;
    V_ID_EMPLEADO    EMPLOYEES.EMPLOYEE_ID%TYPE := &ID_EMPLEADO;
    V_SUELDO         EMPLOYEES.SALARY%TYPE;
    V_SUELDO_NUEVO   NUMBER;
    
    V_AUMENTO        NUMBER := &AUMENTO;
    
    E_AUMENTO_EXCESIVO EXCEPTION;
BEGIN
    SELECT FIRST_NAME, SALARY
    INTO V_NOMBRE, V_SUELDO
    FROM EMPLOYEES
    WHERE EMPLOYEE_ID = V_ID_EMPLEADO;
    
    IF V_AUMENTO > 40 THEN
        RAISE E_AUMENTO_EXCESIVO;
    END IF;
    
    V_SUELDO_NUEVO := (V_SUELDO * V_AUMENTO) / 100;
    
    DBMS_OUTPUT.PUT_LINE('EL NOMBRE DEL EMPLEADO ES: '||V_NOMBRE);
    DBMS_OUTPUT.PUT_LINE('SU SUELDO ES: $'||V_SUELDO);
    DBMS_OUTPUT.PUT_LINE('CON EL AUMENTO QUEDARIA EN: $'||V_SUELDO_NUEVO);
EXCEPTION
    WHEN NO_DATA_FOUND THEN -- NO SE ENCONTRO INFORMACION
        DBMS_OUTPUT.PUT_LINE('EL EMPLEADO CON ID '||V_ID_EMPLEADO||' NO EXISTE');
    WHEN OTHERS THEN -- EN CASO DE CUALQUIER COSA, SIEMPRE PONERLO AL FINAL
        DBMS_OUTPUT.PUT_LINE('OCURRIO UN ERROR INESPERADO');
END;

DECLARE
    E_RESTRICCION EXCEPTION;
    PRAGMA EXCEPTION_INIT(E_RESTRICCION, -02292);
BEGIN
    DELETE FROM DEPARTMENTS
    WHERE DEPARTMENT_ID = 50;
