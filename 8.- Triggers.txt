---------------------------------------------------------------
CREATE OR REPLACE TRIGGER TRG_LOG_EMPLEADOS
AFTER UPDATE OR DELETE OR INSERT ON EMPLEADOS
BEGIN
    IF UPDATING THEN
        INSERT INTO LOGS_EMPLEADO VALUES(
            USER,
            SYSDATE,
            'Se actualizo la tabla empleados'
        );
    ELSIF INSERTING THEN
        INSERT INTO LOGS_EMPLEADO VALUES(
            USER,
            SYSDATE,
            'Se inserto en la tabla empleados'
        );
    ELSIF DELETING THEN
        INSERT INTO LOGS_EMPLEADO VALUES(
            USER,
            SYSDATE,
            'Se elimino de la tabla empleados'
        );
    END IF;
END;

UPDATE EMPLEADOS

CREATE TABLE LOG_SALARY(
    EMPLEADO_ID        NUMBER,
    SALARIO_ANTERIOR   NUMBER,
    SALARIO_NUEVO      NUMBER,
    USUARIO            VARCHAR2(20),
    FECHA              DATE
);

CREATE OR REPLACE TRIGGER TRG_LOG_SALARIO
AFTER UPDATE OF SALARY ON EMPLEADOS
FOR EACH ROW
BEGIN
    INSERT INTO LOG_SALARY VALUES(
        :OLD.EMPLOYEE_ID,
        :OLD.SALARY,
        :NEW.SALARY,
        USER,
        SYSDATE
    );
END;
